═══════════════════════════════════════════════════════════════════════════════
RECOMMENDED: Use Gradient Extensions for EASIEST migration
═══════════════════════════════════════════════════════════════════════════════

The new theme_system.dart includes a BuildContext extension!

Instead of:
  Container(
    decoration: BoxDecoration(
      gradient: TravelloTheme.gradientAccentGold,
      borderRadius: TravelloTheme.radiusMedium,
    ),
    child: child,
  )

Use the shorthand:
  context.gradientContainer(
    gradientName: 'accent',
    child: child,
  )

Or get gradients dynamically:
  LinearGradient myGradient = context.getGradient('hero');

EXAMPLE: Real-world refactoring - Before & After
═══════════════════════════════════════════════════════════════════════════════

BEFORE (Old scattered themes):

import 'package:flight_app/ui/themes/theme_palette.dart';
import 'package:flight_app/ui/themes/theme_text.dart';
import 'package:flight_app/ui/themes/theme_spacing.dart';

class FeatureCard extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: EdgeInsets.all(spacingUnit(2)),
      child: Container(
        decoration: BoxDecoration(
          color: ThemePalette.primaryLight,
          borderRadius: ThemeRadius.medium,
          boxShadow: [
            BoxShadow(
              color: Colors.black.withOpacity(0.1),
              blurRadius: 8,
            )
          ],
        ),
        child: Padding(
          padding: EdgeInsets.all(spacingUnit(2)),
          child: Text('Book Now', style: ThemeText.subtitle),
        ),
      ),
    );
  }
}


AFTER (Unified + Gradient):
────────────────────────────
import 'package:flight_app/ui/themes/theme_system.dart';

class FeatureCard extends StatelessWidget {
  final String type; // 'flight', 'train', 'hotel'

  const FeatureCard({required this.type});

  @override
  Widget build(BuildContext context) {
    final gradientMap = {
      'flight': 'sky',
      'train': 'sunset',
      'hotel': 'ocean',
    };

    return Padding(
      padding: EdgeInsets.all(TravelloTheme.spacing(2)),
      child: context.gradientContainer(
        gradientName: gradientMap[type] ?? 'accent',
        borderRadius: TravelloTheme.radiusMedium,
        shadows: TravelloTheme.shadowMedium,
        child: Padding(
          padding: EdgeInsets.all(TravelloTheme.spacing(2)),
          child: Text('Book Now', style: TravelloTheme.subtitle2),
        ),
      ),
    );
  }
}
